<script>
let audioContext, analyser, dataArray;
let listening = false;
let firstBlowDone = false;
let candlesBlown = false;

// voice message
function speakWish() {
  const msg = new SpeechSynthesisUtterance("Happy Birthday Beba ❤️");
  msg.rate = 0.9;
  speechSynthesis.speak(msg);
}

// wind sound
function playWindSound() {
  const ctx = new (window.AudioContext || window.webkitAudioContext)();
  const noiseBuffer = ctx.createBuffer(1, ctx.sampleRate * 1, ctx.sampleRate);
  const output = noiseBuffer.getChannelData(0);
  for (let i = 0; i < output.length; i++) output[i] = Math.random() * 2 - 1;
  const noise = ctx.createBufferSource();
  noise.buffer = noiseBuffer;
  const gainNode = ctx.createGain();
  gainNode.gain.setValueAtTime(0.3, ctx.currentTime);
  gainNode.gain.exponentialRampToValueAtTime(0.01, ctx.currentTime + 1);
  noise.connect(gainNode);
  gainNode.connect(ctx.destination);
  noise.start();
}

window.onload = () => startMic();

async function startMic() {
  const message = document.getElementById('message');
  try {
    const stream = await navigator.mediaDevices.getUserMedia({ audio: true });
    audioContext = new (window.AudioContext || window.webkitAudioContext)();
    const source = audioContext.createMediaStreamSource(stream);
    analyser = audioContext.createAnalyser();
    analyser.fftSize = 256;
    source.connect(analyser);
    dataArray = new Uint8Array(analyser.frequencyBinCount);
    listening = true;
    message.innerText = "Blow the candles 💨";
    detectBlow();
  } catch {
    message.innerText = "Please allow microphone 😢";
  }
}

function detectBlow() {
  if (!listening) return;
  analyser.getByteFrequencyData(dataArray);
  let volume = dataArray.reduce((a, b) => a + b) / dataArray.length;
  const flames = document.querySelectorAll('.flame');

  if (!firstBlowDone && volume > 45) {
    firstBlowDone = true;
    flames.forEach(f => f.classList.add('strong-flicker'));
    playWindSound();
    document.getElementById('message').innerText = "Blow harder please... ZOOR LAGAO 💨🔥";
    setTimeout(() => requestAnimationFrame(detectBlow), 800);
    return;
  }

  if (firstBlowDone && volume > 60) {
    listening = false;
    playWindSound();
    blowOutCandles();
    return;
  }

  requestAnimationFrame(detectBlow);
}

function blowOutCandles() {
  const flames = document.querySelectorAll('.flame');
  const message = document.getElementById('message');
  message.innerText = "Candles blown! Now cut the cake 🎂";

  setTimeout(() => {
    flames.forEach(f => f.style.display = 'none');
    candlesBlown = true;
    document.body.classList.add('knife-cursor');
    launchConfetti();
  }, 1200);
}

function cutCake() {
  if (!candlesBlown) return;
  const cake = document.getElementById('cake');
  cake.classList.add('cut');
  speakWish();
  sidePopper();
}

// confetti burst
const canvas = document.getElementById('confetti');
const ctx = canvas.getContext('2d');
canvas.width = window.innerWidth;
canvas.height = window.innerHeight;
let pieces = [];
const colors = ["#ff0a54","#ff477e","#00f5d4","#9b5de5","#00bbf9"];

function createConfetti(xStart) {
  for (let i = 0; i < 120; i++) {
    pieces.push({
      x: xStart,
      y: canvas.height / 2,
      r: Math.random() * 6 + 3,
      color: colors[Math.floor(Math.random() * colors.length)],
      angle: Math.random() * 2 * Math.PI,
      speed: Math.random() * 6 + 3
    });
  }
}

function drawConfetti() {
  ctx.clearRect(0, 0, canvas.width, canvas.height);
  pieces.forEach(p => {
    ctx.beginPath();
    ctx.arc(p.x, p.y, p.r, 0, Math.PI * 2);
    ctx.fillStyle = p.color;
    ctx.fill();
  });
  pieces.forEach(p => {
    p.x += Math.cos(p.angle) * p.speed;
    p.y += Math.sin(p.angle) * p.speed + 2;
  });
}

function launchConfetti() {
  createConfetti(canvas.width / 2);
  setInterval(drawConfetti, 20);
}

function sidePopper() {
  createConfetti(0); // left side
  createConfetti(canvas.width); // right side
}
