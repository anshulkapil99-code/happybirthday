    <div class="container">
        <h1>Snake Game</h1>
        <div class="score-board">Score: <span id="scoreVal">0</span></div>
        <canvas id="snakeGame" width="400" height="400"></canvas>
        <div class="controls">Press Arrow Keys to Start</div>
    </div>

    <script>
        const canvas = document.getElementById("snakeGame");
        const ctx = canvas.getContext("2d");
        const scoreElement = document.getElementById("scoreVal");

        const box = 20;
        let score = 0;
        let gameSpeed = 150; // Chill speed

        let snake = [
            { x: 10 * box, y: 10 * box },
            { x: 9 * box, y: 10 * box },
            { x: 8 * box, y: 10 * box }
        ];

        let food = getRandomFood();
        let d = null;

        document.addEventListener("keydown", direction);

        function direction(event) {
            let key = event.keyCode;
            if ((key == 37 || key == 65) && d != "RIGHT") d = "LEFT";
            else if ((key == 38 || key == 87) && d != "DOWN") d = "UP";
            else if ((key == 39 || key == 68) && d != "LEFT") d = "RIGHT";
            else if ((key == 40 || key == 83) && d != "UP") d = "DOWN";
        }

        function getRandomFood() {
            return {
                x: Math.floor(Math.random() * 20) * box,
                y: Math.floor(Math.random() * 20) * box,
                color: `hsl(${Math.random() * 360}, 70%, 60%)`
            };
        }

        function collision(head, array) {
            for (let i = 0; i < array.length; i++) {
                if (head.x == array[i].x && head.y == array[i].y) return true;
            }
            return false;
        }

        function draw() {
            // Dark Background
            ctx.fillStyle = "#000";
            ctx.fillRect(0, 0, canvas.width, canvas.height);

            // Draw Snake
            for (let i = 0; i < snake.length; i++) {
                // Gradient effect from Head to Tail
                ctx.fillStyle = (i == 0) ? "#2ecc71" : `hsl(${180 + (i * 5)}, 80%, 60%)`;
                ctx.shadowBlur = (i == 0) ? 15 : 5;
                ctx.shadowColor = ctx.fillStyle;
                
                // Rounded snake segments
                ctx.beginPath();
                ctx.roundRect(snake[i].x + 1, snake[i].y + 1, box - 2, box - 2, 5);
                ctx.fill();
                ctx.shadowBlur = 0;
            }

            // Draw Food (Glowing Pulse)
            ctx.fillStyle = food.color;
            ctx.shadowBlur = 15;
            ctx.shadowColor = food.color;
            ctx.beginPath();
            ctx.arc(food.x + box/2, food.y + box/2, box/2.5, 0, Math.PI * 2);
            ctx.fill();
            ctx.shadowBlur = 0;

            let snakeX = snake[0].x;
            let snakeY = snake[0].y;

            if (d == "LEFT") snakeX -= box;
            if (d == "UP") snakeY -= box;
            if (d == "RIGHT") snakeX += box;
            if (d == "DOWN") snakeY += box;

            // --- WALL WRAPPING LOGIC ---
            if (snakeX < 0) snakeX = canvas.width - box;
            else if (snakeX >= canvas.width) snakeX = 0;
            
            if (snakeY < 0) snakeY = canvas.height - box;
            else if (snakeY >= canvas.height) snakeY = 0;

            // Check for Food
            if (snakeX == food.x && snakeY == food.y) {
                score++;
                scoreElement.innerHTML = score;
                food = getRandomFood();
            } else {
                if(d) snake.pop(); // Only move if a direction is set
            }

            let newHead = { x: snakeX, y: snakeY };

            // Game Over only on Self-Collision
            if (d && collision(newHead, snake)) {
                clearInterval(game);
                alert("You bit yourself! Final Score: " + score + " ❤️");
                location.reload();
            }

            if(d) snake.unshift(newHead);
        }

        let game = setInterval(draw, gameSpeed);
    </script>
