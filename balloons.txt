<script>
let score=0, missed=0, gameRunning=false;
let lastPop=0, combo=1;

const scoreEl=document.getElementById("score");
const missedEl=document.getElementById("missed");
const bgMusic=document.getElementById("bgMusic");

/* START */
startBtn.onclick=()=>{
    startScreen.style.display="none";
    gameRunning=true;
    bgMusic.volume=0.3;
    bgMusic.play();
};

/* MODE */
function isCakeMode(){
    return Math.floor(score/50)%2===1;
}

/* DAY/NIGHT */
function updateTheme(){
    let phase=Math.floor(score/50)%2;

    if(phase===0){
        document.body.style.background="linear-gradient(to top,#87CEEB,#fff)";
        sun.style.display="block";
        moon.style.display="none";
        document.querySelectorAll(".star").forEach(s=>s.remove());
    } else {
        document.body.style.background="#000";
        sun.style.display="none";
        moon.style.display="block";

        if(!document.querySelector(".star")){
            for(let i=0;i<80;i++){
                let s=document.createElement("div");
                s.className="star";
                s.style.left=Math.random()*window.innerWidth+"px";
                s.style.top=Math.random()*window.innerHeight+"px";
                document.body.appendChild(s);
            }
        }
    }
}

/* SCORE + COMBO */
function updateScore(){
    let now=Date.now();
    combo=(now-lastPop<600)?combo+1:1;
    lastPop=now;

    score+=combo;
    scoreEl.innerText="Score: "+score+" x"+combo;

    updateTheme();
}

/* OBJECT */
function createObject(){
    if(!gameRunning) return;

    let obj=document.createElement("div");

    if(isCakeMode()){
        obj.className="cake";
        let c=document.createElement("div");
        c.className="candle";
        let f=document.createElement("div");
        f.className="flame";
        c.appendChild(f);
        obj.appendChild(c);
    } else {
        obj.className="balloon";
        obj.style.background=["red","blue","green","yellow"][Math.floor(Math.random()*4)];
        let s=document.createElement("div");
        s.className="string";
        obj.appendChild(s);
    }

    obj.style.left=Math.random()*(window.innerWidth-100)+"px";
    obj.style.bottom="0px";
    document.body.appendChild(obj);

    let rise=0;
    let speed=isCakeMode()?6:3;

    let int=setInterval(()=>{
        rise+=speed;
        obj.style.bottom=rise+"px";

        if(rise>window.innerHeight){
            obj.remove();
            clearInterval(int);
            missed++;
            missedEl.innerText="Missed: "+missed;

            if(missed>=10) endGame();
        }
    },20);

    const pop=()=>{
        burstSound.play();
        createParticles(obj);
        updateScore();
        obj.remove();
        clearInterval(int);
    };

    obj.onclick=pop;
    obj.ontouchstart=pop;
}

/* BURST */
function createParticles(el){
    let r=el.getBoundingClientRect();
    for(let i=0;i<10;i++){
        let p=document.createElement("div");
        p.className="particle";
        p.style.left=r.left+"px";
        p.style.top=r.top+"px";
        p.style.background=el.style.background||"pink";
        p.style.setProperty("--x",(Math.random()-0.5)*150+"px");
        p.style.setProperty("--y",(Math.random()-0.5)*150+"px");
        document.body.appendChild(p);
        setTimeout(()=>p.remove(),500);
    }
}

/* GAME OVER */
function endGame(){
    gameRunning=false;
    finalScore.innerText="Score: "+score;
    gameOver.style.display="flex";

    let i=0;
    let fx=setInterval(()=>{
        createFireworks();
        if(++i>6) clearInterval(fx);
    },700);

    setInterval(()=>{
        let s=document.createElement("div");
        s.className="sparkle";
        s.style.left=Math.random()*window.innerWidth+"px";
        s.style.top=Math.random()*window.innerHeight+"px";
        document.body.appendChild(s);
        setTimeout(()=>s.remove(),1000);
    },100);
}

/* FIREWORKS */
function createFireworks(){
    fireSound.play();
    for(let i=0;i<30;i++){
        let f=document.createElement("div");
        f.className="firework";
        f.style.left=Math.random()*window.innerWidth+"px";
        f.style.top=Math.random()*window.innerHeight/2+"px";
        f.style.background=["red","yellow","blue"][Math.floor(Math.random()*3)];
        f.style.setProperty("--x",(Math.random()-0.5)*300+"px");
        f.style.setProperty("--y",(Math.random()-0.5)*300+"px");
        document.body.appendChild(f);
        setTimeout(()=>f.remove(),1000);
    }
}

restartBtn.onclick=()=>location.reload();

setInterval(createObject,700);
