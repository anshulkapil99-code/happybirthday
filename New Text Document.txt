<script>
window.onload = function() {
    if (confirm("YOU ARE NOT READY FOR THIS...")) {
        if (confirm("YOU ARE STILL NOT READY FOR THIS..")) {
            document.getElementById("intro-sequence").style.display = "flex";
            startIntroSequence();
        } else { location.reload(); }
    } else { location.reload(); }
};

function startIntroSequence() {
    const seed = document.getElementById("intro-text-seed");
    const groundHeart = document.getElementById("ground-heart-outline");
    const challenge = document.getElementById("challenge-card");
    seed.style.opacity = 1;
    setTimeout(() => { 
        seed.style.opacity = 0; 
        setTimeout(() => {
            groundHeart.style.opacity = 1;
            setTimeout(() => {
                challenge.style.display = "block";
                setTimeout(() => challenge.style.opacity = 1, 100);
            }, 1000);
        }, 1000);
    }, 2000);
}

function revealBirthday() {
    const music = document.getElementById("bgMusic");
    music.play().catch(e => console.log("Audio blocked."));
    document.getElementById("intro-sequence").style.opacity = 0;
    setTimeout(() => {
        document.getElementById("intro-sequence").style.display = "none";
        document.getElementById("main-content").style.display = "block";
        initSite();
        setTimeout(() => document.getElementById("main-content").style.opacity = 1, 100);
    }, 1500);
}

function revealGift() { 
    document.getElementById("gift-box").style.display = "none";
    document.getElementById("msg").style.display = "block"; 
}

function openSong(url) { 
    document.getElementById("bgMusic").pause(); 
    window.open(url, '_blank'); 
}

function checkPassword() {
    let pw = prompt("HINT: What does Vanshika call you?");
    if (pw && pw.toLowerCase().trim() === "sakhi") {
        window.open('https://www.youtube.com/shorts/Wo4DAtWOq5Y', '_blank');
    } else if (pw !== null) alert("Wrong answer! ❤️");
}

function initSite() {
    const birth = new Date(2001, 3, 26, 16, 7, 0); 
    setInterval(() => {
        const now = new Date();
        let y = now.getFullYear() - birth.getFullYear(), m = now.getMonth() - birth.getMonth(), d = now.getDate() - birth.getDate();
        let hh = now.getHours() - birth.getHours(), mm = now.getMinutes() - birth.getMinutes(), ss = now.getSeconds() - birth.getSeconds();
        if (ss < 0) { ss += 60; mm--; } if (mm < 0) { mm += 60; hh--; } if (hh < 0) { hh += 24; d--; }
        if (d < 0) { d += new Date(now.getFullYear(), now.getMonth(), 0).getDate(); m--; }
        if (m < 0) { m += 12; y--; }
        document.getElementById("yrs").innerText = y; document.getElementById("mon").innerText = m; document.getElementById("d").innerText = d;
        document.getElementById("h").innerText = hh; document.getElementById("m").innerText = mm; document.getElementById("s").innerText = ss;
    }, 1000);

    setInterval(() => {
        let h = document.createElement("div"); h.innerHTML = "❤️"; h.className = "heart-float";
        h.style.left = Math.random()*100+"vw"; h.style.fontSize = Math.random()*10+10+"px";
        h.style.animationDuration = Math.random()*3+3+"s";
        document.body.appendChild(h); setTimeout(() => h.remove(), 6000);
    }, 500);

    const starCont = document.getElementById("stars");
    for(let i=0; i<150; i++){
        let s = document.createElement("div"); s.className = "star";
        s.style.width = s.style.height = Math.random()*3+"px";
        s.style.top = Math.random()*100+"vh"; s.style.left = Math.random()*100+"vw";
        starCont.appendChild(s);
    }
}
